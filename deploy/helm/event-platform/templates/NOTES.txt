  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "event-platform.componentName" (dict "root" . "component" "publisher") }} 8000:{{ .Values.publisher.service.port }}

    -H 'content-type: application/json' \
    -d '{"type":"order.created","payload":{"order_id":"demo-1"}}'
